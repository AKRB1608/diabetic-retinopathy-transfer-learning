Write-Host "`n===== 3. DATASET SPLITS =====" -ForegroundColor Cyan

$Splits = @(
    "outputs\train_split.csv",
    "outputs\validation_split.csv",
    "outputs\test_split.csv"
)

$Total = 0
$AllIds = @()

foreach ($Path in $Splits) {
    if (!(Test-Path $Path)) {
        Add-Check "Dataset" $Path "MISSING" "CSV not found."
        continue
    }

    $Rows = @(Import-Csv $Path)
    $Total += $Rows.Count

    Write-Host "`n$Path : $($Rows.Count) rows"
    $Rows | Group-Object diagnosis |
        Select-Object Name, Count |
        Format-Table -AutoSize

    $BadLabels = @(
        $Rows | Where-Object {
            (([int]$_.diagnosis -ge 2) -ne ([int]$_.label -eq 1))
        }
    ).Count

    Add-Check "Dataset" "$Path row count" "VERIFIED" `
        "$($Rows.Count) rows."

    Add-Check "Dataset" "$Path binary labels" `
        $(if ($BadLabels -eq 0) {"VERIFIED"} else {"INCONSISTENT"}) `
        "$BadLabels labels do not match grades 0-1 => 0 and grades 2-4 => 1."

    $AllIds += @($Rows | ForEach-Object { $_.id_code })
}

Add-Check "Dataset" "Total dataset split count" `
    $(if ($Total -eq 3662) {"VERIFIED"} else {"INCONSISTENT"}) `
    "Total=$Total; expected 3662."

$DuplicateIds = @($AllIds | Group-Object | Where-Object Count -gt 1)

Add-Check "Dataset" "Duplicate IDs across splits" `
    $(if ($DuplicateIds.Count -eq 0) {"VERIFIED"} else {"INCONSISTENT"}) `
    "$($DuplicateIds.Count) IDs occur more than once."

Write-Host "`n===== 5. SAVED PREDICTIONS =====" -ForegroundColor Cyan

$PythonAudit = @'
import json
import numpy as np
import pandas as pd
from sklearn.metrics import confusion_matrix, accuracy_score, precision_score, recall_score, f1_score, roc_auc_score

result = {}
npz_path = "outputs/referable_test_predictions.npz"
csv_path = "outputs/test_split.csv"

with np.load(npz_path, allow_pickle=False) as d:
    result["keys"] = list(d.files)
    y = d["y_true"].astype(int).reshape(-1)
    saved_pred = d["y_pred"].astype(int).reshape(-1)
    p = d["probabilities"].astype(float).reshape(-1)

df = pd.read_csv(csv_path)
csv_labels = df["label"].to_numpy(dtype=int)

result["samples"] = len(y)
result["csv_rows"] = len(csv_labels)
result["labels_match_csv_order"] = bool(np.array_equal(y, csv_labels))
result["saved_predictions_match_threshold_0_50"] = bool(
    np.array_equal(saved_pred, (p >= 0.5).astype(int))
)
result["probabilities_finite"] = bool(np.isfinite(p).all())
result["probability_range"] = [float(np.min(p)), float(np.max(p))]
result["metrics"] = {}

for threshold in (0.50, 0.71):
    pred = (p >= threshold).astype(int)
    tn, fp, fn, tp = confusion_matrix(y, pred, labels=[0, 1]).ravel()
    result["metrics"][str(threshold)] = {
        "TN": int(tn), "FP": int(fp), "FN": int(fn), "TP": int(tp),
        "accuracy": float(accuracy_score(y, pred)),
        "precision": float(precision_score(y, pred, zero_division=0)),
        "recall": float(recall_score(y, pred, zero_division=0)),
        "f1": float(f1_score(y, pred, zero_division=0)),
        "macro_f1": float(f1_score(y, pred, average="macro", zero_division=0)),
        "roc_auc": float(roc_auc_score(y, p))
    }

print(json.dumps(result, indent=2))
'@
